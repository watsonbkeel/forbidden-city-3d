import '../lib/inner-palace-fixture.mjs';
import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';import * as T from 'three';
import {CourtKit} from '../../src/scene/palace/kit.js';
import {realTextureLibrary} from '../lib/r48-real-texture-library.mjs';
import {buildRemainingHall,resolveRemainingProfile} from '../../src/scene/palace/special/RemainingOuterHalls.js';
function fixture(){const lib=realTextureLibrary(),kit=new CourtKit(lib,'outer-west'),item=JSON.parse(fs.readFileSync('data/zones/outer-west.json')).items.find(x=>x.name==='南薰殿'),built=buildRemainingHall(kit,item,resolveRemainingProfile('outer-west',item));kit.buildPlaqueAtlas();kit.root.updateMatrixWorld(true);return {...built,kit,lib};}
function close(h){h.kit.dispose(h.kit.root);h.lib.dispose();}
test('actual Nanxun caisson contains source-bounded coiled body, face and layered brackets rather than an unmodeled disk',()=>{const h=fixture();try{const m=h.f.children.filter(x=>x.userData.r137NanxunRelief);assert.ok(m.length>20,'missing actual relief meshes');for(const role of ['dragon-coiled-body','dragon-face','dragon-muzzle','dragon-horn','caisson-ring-bracket'])assert.ok(m.some(x=>x.userData.role===role),role);assert.equal(h.g.caisson.dragonRelief,'PARTIAL_PHOTO_FIT_NOT_EXACT_CARVING');for(const mesh of m){mesh.geometry.computeBoundingBox();const b=mesh.geometry.boundingBox.clone().applyMatrix4(mesh.matrix);assert.ok(b.min.y>6.5&&b.max.y<8,'bounded ceiling relief');for(const v of mesh.geometry.attributes.position.array)assert.ok(Number.isFinite(v));assert.equal(mesh.material.transparent,false);assert.equal(mesh.material.map,null);assert.equal(mesh.userData.source,'https://img.dpm.org.cn/Uploads/Picture/dc/3938[1024].jpg');}assert.equal(h.g.inventory.status,'UNKNOWN');}finally{close(h);}});
