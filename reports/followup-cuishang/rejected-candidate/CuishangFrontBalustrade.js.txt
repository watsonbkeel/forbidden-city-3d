import * as THREE from 'three';
import {createTriangleTemplate,triangleInstance} from '../../../utils/TriangleSupport.js';
const PHOTO='https://img.dpm.org.cn/Uploads/Picture/dc/1496.jpg';
/** Front four open bays only. Original continuous rails, west dark bay and
 * side/rear barriers remain. Wood identity is textual; weathering is a fit. */
export function fitCuishangFrontBalustrade(c){
 const {p,f,kit}=c;if(p.key!=='cuishang'||p.id!=='place-ningshou-b6cde8615b'||p.localW!==20.5||p.localD!==10.9||p.upperFloorY!==4.3||p.rot!==0||f.position.x!==p.x||f.position.y!==0||f.position.z!==p.z||f.rotation.x||f.rotation.y||f.rotation.z||!f.scale.equals(new THREE.Vector3(1,1,1))||f.userData.cuishangFrontBalustrade)throw Error('Cuishang retained front rail binding required');
 const y=p.upperFloorY,z=-(p.localD-.2)/2,w=p.localW-.2,n=Math.ceil(w/.65),end=p.localW/2-p.localW/5;
 const rods=f.children.filter(m=>m.userData.role==='gallery-rail'&&m.position.z===z&&m.geometry.parameters.width===.07&&m.position.x>-w/2+1e-8&&m.position.x<end).sort((a,b)=>a.position.x-b.position.x);
 if(rods.length!==25)throw Error('Cuishang front rod inventory changed');const expected=new THREE.BoxGeometry(.07,.9,.08),matched=[];
 try{for(let i=0;i<rods.length;i++){const m=rods[i];if(Math.abs(m.position.x-(-w/2+(i+1)*w/n))>1e-10||m.position.y!==y+.45||m.rotation.x||m.rotation.y||m.rotation.z||!m.scale.equals(new THREE.Vector3(1,1,1)))throw Error('Cuishang old rod transform changed');for(const key of['position','normal','uv','index']){const a=key==='index'?m.geometry.index:m.geometry.attributes[key],b=key==='index'?expected.index:expected.attributes[key];if(!a||a.count!==b.count||a.itemSize!==b.itemSize||a.array.some((v,j)=>!Object.is(v,b.array[j])))throw Error('Cuishang old rod ordered geometry changed');}m.updateWorldMatrix(true,false);const box=new THREE.Box3().setFromObject(m),col=kit.colliders.find(q=>!matched.includes(q)&&q.min.distanceTo(box.min)<1e-6&&q.max.distanceTo(box.max)<1e-6);if(!col)throw Error('Cuishang old rod collider missing');matched.push(col);}}finally{expected.dispose();}
 const cap=f.children.find(m=>m.userData.role==='gallery-rail'&&m.position.z===z&&m.position.x===0&&m.position.y===y+.88&&m.geometry.parameters.width===w);
 if(!cap)throw Error('Cuishang original continuous cap missing');cap.updateWorldMatrix(true,false);const capBox=new THREE.Box3().setFromObject(cap),capCol=kit.colliders.find(q=>q.min.distanceTo(capBox.min)<1e-6&&q.max.distanceTo(capBox.max)<1e-6);if(!capCol)throw Error('Cuishang original cap contact missing');
 const capExpected=new THREE.BoxGeometry(w,.065,.08);try{if(cap.rotation.x||cap.rotation.y||cap.rotation.z||!cap.scale.equals(new THREE.Vector3(1,1,1)))throw Error('Cuishang original cap transform changed');for(const key of['position','normal','uv','index']){const a=key==='index'?cap.geometry.index:cap.geometry.attributes[key],b=key==='index'?capExpected.index:capExpected.attributes[key];if(!a||a.count!==b.count||a.itemSize!==b.itemSize||a.array.some((v,j)=>!Object.is(v,b.array[j])))throw Error('Cuishang original cap ordered geometry changed');}}finally{capExpected.dispose();}
 for(const m of rods){m.removeFromParent();m.geometry.dispose();}kit.colliders=kit.colliders.filter(q=>!matched.includes(q));
 // The retained cap must participate in the same exact triangle sweep as
 // the new waisted bodies below it; otherwise a falling body can select the
 // lower new support while crossing the older box cap.

 capCol.triangleSupport=triangleInstance(createTriangleTemplate(cap.geometry),[p.x,y+.88,p.z+z],'cuishang-retained-front-cap');capCol.copy(capCol.triangleSupport.box);capCol.cuishangRetainedCap=true;
 const wood=new THREE.MeshStandardMaterial({name:'cuishang-weathered-painted-wood-fit',color:0x8b8c78,roughness:.85});kit.owned.push(wood);
 const box=(role,ww,hh,dd,x,yy)=>{const m=c.tag(kit.mesh(f,new THREE.BoxGeometry(ww,hh,dd),wood,x,yy,z),role);m.userData.cuishangFrontRail=true;m.updateWorldMatrix(true,false);const col=new THREE.Box3().setFromObject(m);col.cuishangFrontRail=true;kit.colliders.push(col);return m;};
 const positions=[-w/2,...rods.map(m=>m.position.x),end];
 for(let i=0;i<positions.length-1;i++){const a=positions[i],b=positions[i+1],ww=b-a-.07,x=(a+b)/2;box('cuishang-lower-framed-panel',ww,.16,.065,x,y+.115);for(const yy of[.035,.195])box('cuishang-lower-panel-frame',ww,.03,.08,x,y+yy);for(const sign of[-1,1])box('cuishang-lower-panel-frame',.03,.16,.08,x+sign*(ww-.03)/2,y+.115);}
 const shape=new THREE.Shape();shape.moveTo(-.09,0);shape.lineTo(.09,0);shape.quadraticCurveTo(.12,.05,.08,.12);shape.quadraticCurveTo(.045,.22,.05,.31);shape.quadraticCurveTo(.065,.36,.12,.39);shape.quadraticCurveTo(.14,.43,.105,.47);shape.lineTo(.065,.52);shape.lineTo(-.065,.52);shape.lineTo(-.105,.47);shape.quadraticCurveTo(-.14,.43,-.12,.39);shape.quadraticCurveTo(-.065,.36,-.05,.31);shape.quadraticCurveTo(-.045,.22,-.08,.12);shape.quadraticCurveTo(-.12,.05,-.09,0);shape.closePath();
 const geometry=new THREE.ExtrudeGeometry(shape,{depth:.08,steps:1,curveSegments:5,bevelEnabled:false});geometry.scale(1,.575/.52,1);geometry.translate(0,0,-.04);const template=createTriangleTemplate(geometry);
 for(let i=0;i<rods.length;i++){const x=rods[i].position.x;box('cuishang-lower-panel-divider',.07,.2075,.08,x,y+.10375);const m=c.tag(kit.mesh(f,geometry.clone(),wood,x,y+.2725,z),'cuishang-waisted-timber-baluster');m.userData.cuishangFrontRail=true;const support=triangleInstance(template,[p.x+x,y+.2725,p.z+z],`cuishang-front-waisted:${i}`),col=support.box.clone();col.triangleSupport=support;col.cuishangFrontRail=true;kit.colliders.push(col);m.userData.contact='SAME_CLOSED_RENDER_TRIANGLES';}geometry.dispose();
 return f.userData.cuishangFrontBalustrade={originalObjectId:'FC-B6CDE8615B',source:PHOTO,textSource:'https://www.dpm.org.cn/explore/building/236433.html',scope:'FRONT_EAST_FOUR_OPEN_GALLERY_BAYS_WEST_DARK_BAY_UNCHANGED',removedPlainRods:rods.length,waistedBalusters:rods.length,lowerPanels:positions.length-1,material:'SOURCE_TEXT_WOOD_WEATHERED_COLOUR_FIT_NOT_STONE',dimensions:'RETAINED_MODEL_ENVELOPE_NOT_SURVEY',ornament:'SIMPLIFIED_WAISTED_OUTLINE_NOT_EXACT_CARVING',counts:'MODEL_SPACING_NOT_PHOTO_COUNT',wholeObjectAccepted:false,remaining:['Fine carved patterns and painted decoration','West dark bay, side and rear detail independently unverified','Complete building/bridge/stair/site acceptance']};
}
