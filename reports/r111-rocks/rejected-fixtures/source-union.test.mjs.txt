import test from 'node:test';import assert from 'node:assert/strict';import * as T from 'three';import {buildDuixiuOuterRock}from'../../src/scene/garden/DuixiuOuterRock.js';import{proveSupportRockUnion}from'../../src/scene/garden/DuixiuSupportRock.js';
const b=buildDuixiuOuterRock({center:new T.Vector3(-29.1,0,509.1),radiusX:12.545,radiusZ:8.305,topY:10,tile:3,supportRock:false});
test('actual 168 disjoint cells exactly cover one retained 10.5 platform solid',()=>{const p=proveSupportRockUnion(b);assert.equal(p.cells.length,168);assert.equal(p.owner.triangleSupport.template.triangles.length,12);assert.equal(p.gridCells,168);assert.equal(p.y0,0);assert.equal(p.y1,10.5);assert.equal(p.capTriangles,672);});
test('incomplete or crossing source fails before mutation',()=>{const p=proveSupportRockUnion(b),old=b.geometries;assert.throws(()=>proveSupportRockUnion({...b,geometries:old.filter((_,i)=>i!==p.cells[0].index)}),/complete|missing/i);const g=p.cells[0].g.clone();g.translate(-.01,0,0);assert.throws(()=>proveSupportRockUnion({...b,geometries:old.map((a,i)=>i===p.cells[0].index?g:a)}),/cross|complete|partition/i);g.dispose();assert.equal(b.geometries,old);});
test.after(()=>b.geometries.forEach(g=>g.dispose()));
