import test from'node:test';import assert from'node:assert/strict';import*as T from'three';import{buildDuixiuOuterRock}from'../../src/scene/garden/DuixiuOuterRock.js';import{proveSupportRockUnion}from'../../src/scene/garden/DuixiuSupportRock.js';
const o={center:new T.Vector3(-29.1,0,509.1),radiusX:12.545,radiusZ:8.305,topY:10,tile:3},old=buildDuixiuOuterRock({...o,supportRock:false}),candidate=buildDuixiuOuterRock(o),proof=proveSupportRockUnion(old);
const attrs=g=>Object.fromEntries(Object.entries(g.attributes).map(([k,a])=>[k,Array.from(a.array)]));
function caps(gs){const rows=[];for(const g of gs){const p=g.attributes.position,ix=g.index;for(let i=0;i<(ix?.count??p.count);i+=3){const ids=[0,1,2].map(k=>ix?ix.getX(i+k):i+k);if(![proof.y0,proof.y1].some(y=>ids.every(k=>p.getY(k)===y)))continue;rows.push(JSON.stringify(Object.fromEntries(Object.entries(g.attributes).map(([n,a])=>[n,ids.flatMap(k=>Array.from(a.array.slice(k*a.itemSize,(k+1)*a.itemSize)))]))));}}return rows.sort();}
test('bounded folded owner replaces planar source with <=160 triangles and shared Float32 surface',()=>{const m=candidate.profile.supportRock;assert.ok(m,'candidate was applied');assert.ok(m.physicsTriangles<=160);assert.ok(m.physicsTriangles>12);const g=candidate.geometries.find(g=>g.userData.naturalOwner===m.owner),c=candidate.colliders.find(c=>c.triangleSupport.id===m.owner);assert.ok(g&&c);const p=g.attributes.position,points=new Set(Array.from({length:p.count},(_,i)=>[p.getX(i),p.getY(i),p.getZ(i)].join(',')));for(const q of c.triangleSupport.template.triangles)for(const v of[q.triangle.a,q.triangle.b,q.triangle.c])assert.ok(points.has(v.toArray().join(',')));assert.ok(Array.from({length:p.count},(_,i)=>p.getX(i)).some(x=>x>proof.owner.min.x&&x<proof.owner.min.x+.7));});
test('every source cap attribute and every unselected geometry/owner remains exact',()=>{const m=candidate.profile.supportRock;assert.ok(m);const g=candidate.geometries.find(g=>g.userData.naturalOwner===m.owner);assert.deepEqual(caps([g]),caps(proof.cells.map(c=>c.g)));const removed=new Set(proof.cells.map(c=>c.index));const keep=old.geometries.filter((_,i)=>!removed.has(i));assert.deepEqual(candidate.geometries.filter(x=>x!==g).map(attrs),keep.map(attrs));const sig=c=>({id:c.triangleSupport.id,offset:c.triangleSupport.offset.toArray(),v:c.triangleSupport.template.triangles.map(q=>[q.triangle.a.toArray(),q.triangle.b.toArray(),q.triangle.c.toArray()])});assert.deepEqual(candidate.colliders.filter(c=>c.triangleSupport.id!==m.owner).map(sig),old.colliders.filter(c=>c!==proof.owner).map(sig));});
test.after(()=>[old,candidate].forEach(b=>b.geometries.forEach(g=>g.dispose())));
