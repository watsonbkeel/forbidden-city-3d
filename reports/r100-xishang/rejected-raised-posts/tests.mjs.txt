import test from'node:test';import assert from'node:assert/strict';import fs from'node:fs';import{gunzipSync}from'node:zlib';import{makeXishang,snapshot,isRail,box,THREE}from'./lib/r100-xishang-fixture.mjs';
const old=JSON.parse(gunzipSync(fs.readFileSync(new URL('./fixtures/r100/xishang-before.json.gz',import.meta.url))));
test('R100 replaces only the two thin open side rails with eight solid stone panels and ten broad posts',()=>{const h=makeXishang();try{const ms=[];h.f.traverse(m=>{if(m.isMesh&&isRail(m))ms.push(m);});assert.equal(ms.filter(m=>m.userData.role==='bamboo-stone-panel').length,8);assert.equal(ms.filter(m=>m.userData.role==='bamboo-stone-post').length,10);assert.equal(ms.filter(m=>m.userData.role==='bamboo-stone-rail').length,0);for(const m of ms.filter(m=>m.userData.role==='bamboo-stone-panel')){m.geometry.computeBoundingBox();const b=m.geometry.boundingBox;assert.ok(b.max.y-b.min.y>.5);assert.ok(b.max.z-b.min.z>.9);}}finally{h.kit.dispose(h.f);}});
test('R100 every rail collider is the exact world envelope of its own visible box, never a hidden span',()=>{const h=makeXishang();try{const ms=[];h.f.traverse(m=>{if(m.isMesh&&isRail(m))ms.push(m);});assert.ok(ms.some(m=>m.userData.role==='bamboo-stone-panel'));assert.equal(ms.length,h.railIndices.length);for(let i=0;i<ms.length;i++){assert.deepEqual(box(new THREE.Box3().setFromObject(ms[i])),box(h.kit.colliders[h.railIndices[i]]));}const envelope=new THREE.Box3();for(const b of old.railColliders)envelope.union(new THREE.Box3(new THREE.Vector3(...b[0]),new THREE.Vector3(...b[1])));for(const m of ms){const b=new THREE.Box3().setFromObject(m);for(const k of['x','y','z']){assert.ok(b.min[k]>=envelope.min[k]-1e-7);assert.ok(b.max[k]<=envelope.max[k]+1e-7);}}}finally{h.kit.dispose(h.f);}});
test('R100 preserves every non-rail attribute, collider, support, map and actual channel sample exactly',()=>{const h=makeXishang();try{const now=JSON.parse(JSON.stringify(snapshot(h)));for(const k of['meshes','colliders','walkables','map','rooms','portals','channel'])assert.deepEqual(now[k],old[k],k);}finally{h.kit.dispose(h.f);}});

test('R100 visible posts stand above the panel cap as in both source views',()=>{const h=makeXishang();try{const ys={};h.f.traverse(m=>{if(!m.isMesh||!isRail(m))return;m.geometry.computeBoundingBox();(ys[m.userData.role]??=[]).push(m.position.y+m.geometry.boundingBox.max.y);});assert.ok(Math.min(...ys['bamboo-stone-post'])-Math.max(...ys['bamboo-stone-cap'])>.1);}finally{h.kit.dispose(h.f);}});
