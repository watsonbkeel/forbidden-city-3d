 // independently checked below by an exact triangle/material multiset.

 const original=W.scene,view={updateMatrixWorld:(...a)=>original.updateMatrixWorld(...a),traverse(fn){original.traverse(m=>{if(m.isMesh&&m.material?.name?.startsWith('r163_gaoyun_'))return;fn(m);});}};

export function hostTriangles(scene,{normalization=null}={}){scene.updateMatrixWorld(true);const rows=[];scene.traverse(m=>{if(!m.isMesh||!m.geometry?.attributes.position)return;let mat=m.material;if(mat?.name?.startsWith('r163_gaoyun_view_glass_')){if(!normalization?.[mat.name])throw Error('Exact window-only material normalization missing');mat=normalization[mat.name];}else if(!mat?.name?.startsWith('r163_jinzhao_rear_'))return;const info=mat.isMaterial?fields(mat):mat,g=m.geometry,a=g.attributes.position,ix=g.index;for(let i=0;i<(ix?.count??a.count);i+=3){const vertex=[];for(let j=0;j<3;j++){const k=ix?.getX(i+j)??i+j,v=new T.Vector3().fromBufferAttribute(a,k).applyMatrix4(m.matrixWorld);vertex.push({position:v.toArray().map(Math.fround),attributes:Object.fromEntries(Object.entries(g.attributes).filter(([n])=>n!=='position').sort(([a],[b])=>a.localeCompare(b)).map(([n,q])=>[n,Array.from(q.array.slice(k*q.itemSize,(k+1)*q.itemSize))]))});}rows.push(JSON.stringify({material:info,vertex}));}});return rows.sort();}
