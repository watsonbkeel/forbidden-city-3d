import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';
import{fixture,meshes,meshRow,hash,THREE}from'./lib/r66-h/fixture.mjs';
const base=JSON.parse(fs.readFileSync(new URL('./fixtures/r66-h/yanchun-front-retention.json',import.meta.url))),old=JSON.parse(fs.readFileSync(new URL('./fixtures/r66-h/yanchun-before.json',import.meta.url))),prefix='r66-yanchun-south-';
test('R66 Yanchun finished south photo: plaque mounted on front beam, not low on recessed wall',()=>{const w=fixture('延春阁');try{const p=w.kit.plaques.find(p=>p.text==='延春阁');assert.ok(p.mesh.position.y>4.8);assert.ok(p.mesh.position.z<-12);assert.equal(p.mesh.position.y,p.frame.position.y);assert.ok(Math.abs(p.frame.position.z-p.mesh.position.z-.06)<1e-8);assert.equal(p.mesh.geometry.parameters.width,.9);assert.equal(p.mesh.geometry.parameters.height,1.5);const target=p.mesh.getWorldPosition(new THREE.Vector3()),eye=new THREE.Vector3(216,2.45,460.5),ray=new THREE.Raycaster(eye,target.clone().sub(eye).normalize());const mat=new THREE.MeshBasicMaterial({side:THREE.DoubleSide});const probes=meshes(w.f).map(m=>{const p=new THREE.Mesh(m.geometry,mat);p.matrixWorld.copy(m.matrixWorld);p.userData.original=m;return p;});const hits=ray.intersectObjects(probes,false).filter(h=>h.distance<=eye.distanceTo(target)+.05);assert.equal(hits[0]?.object.userData.original,p.mesh,'actual eye ray first meets face, not roof/beam');mat.dispose();}finally{w.dispose()}});
test('R66 Yanchun exact south turned balusters and uncarved scallop fascia replace only south bars',()=>{const w=fixture('延春阁');try{const a=meshes(w.f).filter(m=>m.userData.role?.startsWith(prefix));assert.ok(a.length>=28);assert.equal(a.filter(m=>m.userData.role===prefix+'turned-baluster').length,25);assert.equal(a.filter(m=>m.userData.role===prefix+'plain-hanging-fascia').length,1);const pls=new Set(w.kit.plaques.flatMap(p=>[p.mesh,p.frame])),rest=meshes(w.f).filter(m=>!a.includes(m)&&!pls.has(m));assert.equal(rest.length,base.count);assert.equal(hash(rest.map(meshRow)),base.retainedMeshes);for(const k of['walkables','rooms','maps','regions'])assert.equal(hash(w.kit[k==='maps'?'mapItems':k]),old[k]);}finally{w.dispose()}});
test('R66 Yanchun visible railing only: exact proxies, retained collision order, no invisible solid wall',()=>{const w=fixture('延春阁');try{const indices=w.f.userData.r66YanchunSouth?.colliderIndices;assert.ok(Array.isArray(indices));assert.equal(indices.length,27);assert.deepEqual(JSON.parse(JSON.stringify(w.kit.colliders.filter((_,i)=>!indices.includes(i)))),base.retainedColliders);const solids=meshes(w.f).filter(m=>m.userData.r66Collider!==undefined);assert.equal(solids.length,27);for(const m of solids){const b=new THREE.Box3().setFromObject(m),c=w.kit.colliders[m.userData.r66Collider];assert.ok(b.min.distanceTo(c.min)<1e-5&&b.max.distanceTo(c.max)<1e-5);}for(const c of indices.map(i=>w.kit.colliders[i]))assert.ok(c.max.y-c.min.y<.95);}finally{w.dispose()}});

test('R66 south fascia has real bearing contact; lathe surfaces stay finite/nondegenerate',()=>{const w=fixture('延春阁');try{const a=meshes(w.f).filter(m=>m.userData.role?.startsWith(prefix)),fascia=a.find(m=>m.userData.role===prefix+'plain-hanging-fascia'),ring=meshes(w.f).find(m=>m.userData.role==='yanchun-lower-roof-balcony-junction'&&m.position.z<0),bb=new THREE.Box3().setFromObject(ring);let contact=0;for(const m of a){const p=m.geometry.attributes.position,ix=m.geometry.index;for(let k=0;k<(ix?.count??p.count);k+=3){const t=new THREE.Triangle(...[0,1,2].map(j=>new THREE.Vector3().fromBufferAttribute(p,ix?ix.getX(k+j):k+j).applyMatrix4(m.matrixWorld)));assert.ok(t.getArea()>1e-12);if(m===fascia&&bb.intersectsTriangle(t))contact++;}}assert.ok(contact>0);assert.equal(w.f.userData.r66YanchunSouth.entityId,'FC-12BEB1BB0E');}finally{w.dispose()}});

test('R66 exact host and duplicate checks fail before any new allocation',async()=>{const {buildYanchunSouthBalcony}=await import('../src/scene/palace/special/YanchunSouthPhotoFit.js');const w=fixture('延春阁');try{const context={kit:w.kit,f:w.f,p:w.p,g:w.g,item:w.item},count=w.kit.owned.length,n=meshes(w.f).length;assert.throws(()=>buildYanchunSouthBalcony(context),/already/);assert.throws(()=>buildYanchunSouthBalcony({...context,item:{...w.item,x:215.9}}),/exact/);assert.equal(w.kit.owned.length,count);assert.equal(meshes(w.f).length,n);}finally{w.dispose()}});
