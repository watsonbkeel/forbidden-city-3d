import test from'node:test';import assert from'node:assert/strict';import fs from'node:fs';import{fixture,meshes,meshRow,hash}from'./lib/r66-h/fixture.mjs';
const baseline=JSON.parse(fs.readFileSync(new URL('./fixtures/r66-h/inventory-before.json',import.meta.url)));
for(const name of['重华宫','延春阁'])test('R66 exact unknown furniture withheld, not historically empty: '+name,()=>{const w=fixture(name),b=baseline[name];try{assert.equal(meshes(w.f).filter(m=>m.userData.role===b.role).length,0);const x=w.f.userData.structure.inventory;assert.equal(x.status,'UNKNOWN');assert.equal(x.knownEmpty,false);assert.equal(x.accepted,false);assert.equal(x.removedMeshCount,b.removedRows.length);assert.equal(x.removedColliderCount,b.removedColliders.length);assert.equal(hash(meshes(w.f).map(meshRow)),b.retainedMeshesHash);assert.equal(meshes(w.f).length,b.retainedMeshesCount);assert.deepEqual(JSON.parse(JSON.stringify(w.kit.colliders)),b.retainedColliders);for(const k of['walkables','maps'])assert.equal(hash(w.kit[k==='maps'?'mapItems':k]),b[k]);for(const room of w.kit.rooms){assert.equal(room.inventory.knownEmpty,false);assert.equal(room.layout.status,'UNVERIFIED_RETAINED_MODEL_BOUNDS');}const clean=rs=>rs.map(({inventory,layout,...r})=>r);assert.deepEqual(JSON.parse(JSON.stringify(clean(w.kit.rooms))),b.rooms);assert.deepEqual(JSON.parse(JSON.stringify(w.kit.regions)),b.regions);}finally{w.dispose()}});

import {withholdHUnknownInventory}from'../src/scene/palace/special/HUnknownInventory.js';
for(const name of['重华宫','延春阁'])test('R66 inventory duplicate/foreign identity fails without mutations: '+name,()=>{const w=fixture(name);try{const before={meshes:hash(meshes(w.f).map(meshRow)),colliders:hash(w.kit.colliders),rooms:hash(w.kit.rooms)},owned=w.kit.owned.length;assert.throws(()=>withholdHUnknownInventory(w.kit,w.item,w.f),/duplicate/);assert.throws(()=>withholdHUnknownInventory(w.kit,{...w.item,name:'其他殿'},w.f),/host/);assert.equal(w.kit.owned.length,owned);assert.deepEqual({meshes:hash(meshes(w.f).map(meshRow)),colliders:hash(w.kit.colliders),rooms:hash(w.kit.rooms)},before);}finally{w.dispose()}});
import{CourtKit}from'../src/scene/palace/kit.js';import{library}from'./lib/inner-palace-fixture.mjs';
for(const corruption of['vertex','material','collider'])test('R66 exact legacy-removal guard rejects '+corruption+' drift atomically',()=>{const item=JSON.parse(fs.readFileSync(new URL('../data/zones/west-six.json',import.meta.url))).items.find(i=>i.kind==='hall'&&i.name==='重华宫'),kit=new CourtKit(library(),'west-six'),{f}=kit.hall({...item,bays:5});try{const a=meshes(f).filter(m=>m.userData.role==='retained-throne-component');for(const m of a)if(m.geometry.parameters.width>3)m.geometry.scale(.74,1,1);const c=kit.colliders.find(c=>Math.abs(c.min.x-item.x+2.5)<1e-8&&Math.abs(c.min.z-item.z-.52)<1e-8);assert.ok(c);c.min.x=item.x-1.85;c.max.x=item.x+1.85;if(corruption==='vertex')a[0].geometry.attributes.position.array[0]+=.01;if(corruption==='material')a[0].material.color.setHex(0x123456);if(corruption==='collider')c.min.z+=.01;f.updateMatrixWorld(true);const before={meshes:hash(meshes(f).map(meshRow)),colliders:hash(kit.colliders),rooms:hash(kit.rooms)};assert.throws(()=>withholdHUnknownInventory(kit,item,f),/drift/);assert.deepEqual({meshes:hash(meshes(f).map(meshRow)),colliders:hash(kit.colliders),rooms:hash(kit.rooms)},before);}finally{kit.dispose(kit.root)}});
