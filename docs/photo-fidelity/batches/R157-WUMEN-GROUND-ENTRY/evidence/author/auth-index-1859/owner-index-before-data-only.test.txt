import test from 'node:test';import assert from 'node:assert/strict';import * as T from 'three';import {CollisionManager} from '../../../../src/utils/Collision.js';import {registerWumenJointAscentOwners,validateWumenJointAscentBinding} from '../../../../src/controls/WumenJointAscentContact.js';import {buildWumenAscentContactRig} from '../../../../scripts/photo-fidelity/batches/wumen-joint-layout/ascent-contact.mjs';
function setup(t){const mat=new T.MeshStandardMaterial(),rig=buildWumenAscentContactRig(mat),cm=new CollisionManager();for(const p of rig.pieces){cm.addColliders(p.colliders);cm.addWalkable(p.walkable);}t.after(()=>{rig.dispose();mat.dispose();});return{cm,rig};}
test('each full authentication reads unrelated physical owner keys only linearly, without cross-call cache',t=>{const{cm,rig}=setup(t);let reads=0;const n=1024;for(let i=0;i<n;i++){cm.addCollider(new T.Box3(new T.Vector3(1000+i*2,0,1000),new T.Vector3(1001+i*2,1,1001)));const row=cm.colliders.at(-1),mesh=row.mesh;Object.defineProperty(row,'mesh',{enumerable:true,configurable:true,get(){reads++;return mesh;}});}const binding=registerWumenJointAscentOwners(cm,rig.pieces);t.after(()=>binding.dispose());assert.ok(reads<=2*n,`registration owner reads ${reads} exceed linear bound ${2*n}`);reads=0;assert.equal(validateWumenJointAscentBinding(cm),binding);assert.ok(reads>=n&&reads<=2*n,`live revalidation must freshly read each owner, got ${reads}`);});
test('the temporary index preserves duplicate, removed and retargeted physical row rejection',t=>{for(const mutate of[(cm,p)=>cm.colliders.push(cm.colliders.find(c=>c.mesh===p.colliders[0])),cm=>cm.colliders.shift(),(cm,p)=>{cm.colliders.find(c=>c.mesh===p.colliders[0]).mesh=new T.Box3();}]){const{cm,rig}=setup(t),b=registerWumenJointAscentOwners(cm,rig.pieces);mutate(cm,rig.pieces[0]);assert.throws(()=>validateWumenJointAscentBinding(cm),/Wumen joint/);b.dispose();}});
test('physical collection membership changing during owner authentication cannot publish a binding',t=>{const{cm,rig}=setup(t);cm.addCollider(new T.Box3(new T.Vector3(1000,0,1000),new T.Vector3(1001,1,1001)));const mesh=rig.pieces[0].mesh,g=mesh.geometry;let armed=true;Object.defineProperty(mesh,'geometry',{configurable:true,enumerable:true,get(){if(armed){armed=false;cm.colliders.pop();}return g;}});assert.throws(()=>registerWumenJointAscentOwners(cm,rig.pieces),/Wumen joint/);assert.equal(validateWumenJointAscentBinding(cm),null);});
